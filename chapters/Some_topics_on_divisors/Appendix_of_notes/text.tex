\subsection{Fibrations}

\begin{construction}[Relative normalization]
	Let $X$ be a noetherian integral scheme and $\calA$ be a quasi-coherent $\calO_X$-algebra.
	The \emph{integral closure} of $\calO_X$ in $\calA$ is defined as the quasi-coherent $\calO_X$-algebra $\calA'$ such that for every affine open subset $U = \Spec A$ of $X$, we have $\calA'(U) = \overline{A}$, where $\overline{A}$ is the integral closure of $A$ in $\calA(U)$.
	The integral closure exists; see \cite[\href{https://stacks.math.columbia.edu/tag/035F}{Tag 035F}]{Stacks}.

	Let $f: Y \to X$ be a morphism between noetherian integral schemes.
	The \emph{relative normalization of $X$ in $Y$} is defined as
	\[ X' = \calSpec_X \calA', \]
	together with the factorization $Y \xrightarrow{f'} X' \xrightarrow{\nu} X$ of $f$,
	where $\calA'$ is the integral closure of $\calO_X$ in $f_* \calO_Y$.
\end{construction}

In our setting, the relative normalization of $X$ in $Y$ is characterized by the following universal property.

\begin{proposition}\label{prop:universal_property_of_relative_normalization}
	Let $f: Y \to X$ be a morphism between noetherian integral schemes and $Y \xrightarrow{f'} X' \xrightarrow{\nu} X$ be the relative normalization of $X$ in $Y$.
	Assuming that $X,Y$ are excellent and $f$ is of finite type, then we have the following properties.
	\begin{enumerate}
		\item The structure morphism $\nu: X' \to X$ is finite; see \cite[\href{https://stacks.math.columbia.edu/tag/03GR}{Tag 03GR}]{Stacks}.
		\item For every factorization $Y \xrightarrow{f''} X'' \xrightarrow{\pi'} X$ of $f$ such that $\pi'$ is finite and $X''$ is integral, there is a unique morphism $g: X' \to X''$ making the following diagram commute:
		      \[
			      \begin{tikzcd}
				      Y \ar[r, "f'"] \ar[d, "f''"'] & X' \ar[d, "\nu"] \ar[ld, gray, "g"'] \\
				      X'' \ar[r, "\nu'"'] & X
			      \end{tikzcd}
		      \]
		      see \cite[\href{https://stacks.math.columbia.edu/tag/035I}{Tag 035I}]{Stacks}.
	\end{enumerate}
\end{proposition}

\begin{example}
	Let
	\begin{itemize}[nosep]
		\item $f_1$ be a surjective morphism with geometrically integral fibers, for example, $f_1: \bbA^2_\kk \to \bbA^1_\kk$ defined by $\kk[x] \to \kk[x,y]$;
		\item $f_2$ be an open immersion into a normal variety, for example, $f_2: \bbA^1_\kk \injmap \bbP^1_\kk$;
		\item $f_3$ be a finite surjective morphism, for example, $f_3: \bbP^1_\kk \to \bbP^1_\kk$ defined by $[u:v] \mapsto [u^2:v^2]$;
		\item $f_4$ be a closed immersion, for example, $f_4: \bbP^1_\kk \to \bbP^2_\kk$ defined by $[u:v] \mapsto [u^2:uv:v^2]$.
	\end{itemize}
	Set $f = f_4 \circ f_3 \circ f_2 \circ f_1: \bbA^2_\kk \to \bbP^2_\kk$.
	Then the relative normalization of $\bbP^2_\kk$ in $\bbA^2_\kk$ is $\bbP^1_\kk \xrightarrow{f_3} \bbP^1_\kk \xrightarrow{f_4} \bbP^2_\kk$, and the morphism $\bbA^2_\kk \to \bbP^1_\kk$ is given by $\bbA^2_\kk \xrightarrow{f_1} \bbA^1_\kk \xrightarrow{f_2} \bbP^1_\kk$.
\end{example}

\begin{example}
	Let $X$ be an integral excellent noetherian scheme and $U$ be the normal locus of $X$.
	Then the relative normalization of $X$ in $U$ is $U \injmap X' \xrightarrow{\nu} X$, where $\nu$ is the normalization of $X$.
\end{example}

By the construction of the relative normalization, we can factor a morphism $f: Y \to X$ into a ``finite part'' $X' \to X$ and another morphism $Y \to X'$.
Now we describe the part $Y \to X'$ in more details, which motivates the definition of fibrations.

\begin{definition}
	Let $f: X \to Y$ be a morphism of finite type between noetherian integral schemes.
	We say $f$ is a \emph{fibration} if the relative normalization of $Y$ in $X$ is identical to $Y$, i.e. the structure morphism $Y' \to Y$ is an isomorphism.
	%  \Yang{}
\end{definition}

\begin{proposition}\label{prop:basic_properties_of_fibrations}
	Let $f: X \to Y$ be a morphism of finite type between excellent noetherian integral schemes.
	\begin{enumerate}
		\item If $f$ is a fibration, then $f$ is dominant.
		\item Let $Y' \to Y$ be the relative normalization of $Y$ in $X$, then the morphism $X \to Y'$ is a fibration.
		\item The relative normalization $Y'$ of $Y$ in $X$ is still excellent noetherian integral.
		      In particular, if $X$ and $Y$ are varieties over $\kk$, then the relative normalization of $Y$ in $X$ is also a variety over $\kk$.
	\end{enumerate}
\end{proposition}
\begin{proof}
	For (a), let $Z$ be the scheme-theoretic image of $f$.
	Then $f$ factors through the closed immersion $Z \to Y$ and hence $Y \xrightarrow{\id} Y$ factors through $Z$ by the universal property of the relative normalization, which implies $Z = Y$ and hence $f$ is dominant.

	Assertion (b) follows from the universal property of the relative normalization (\cref{prop:universal_property_of_relative_normalization}).

	Now we prove (c).
	Denote by $g:Y' \to Y$ the relative normalization of $Y$ in $X$.
	Since $g$ is finite, $Y'$ is excellent and noetherian.
	Since $X \to Y'$ is dominant and $X$ is irreducible, $Y'$ is also irreducible.
	Take any affine open subset $U \subset Y$, $g^{-1}(U)$ is still affine.
	We have $\calO_{Y'}(g^{-1}(U)) \subset \calO_X(f^{-1}(U))$.
	Since $X$ is integral, $\calO_{Y'}(g^{-1}(U))$ is reduced.
	This shows that $Y'$ is reduced.

	If $X$ and $Y$ are varieties over $\kk$, we only need to check that $Y'$ is geometrically integral over $\kk$.
	Note that $\calO_{Y'_\kkk}$ locally is still a subalgebra of $f_* \calO_{X_\kkk}$ and dominance is preserved under flat base change (\cite[\href{https://stacks.math.columbia.edu/tag/0H3F}{Tag 0H3F}]{Stacks}), so $Y'_\kkk$ is still integral.
\end{proof}

There may be not a standard definition of fibration in literatures, but the following proposition shows that the above definition is equivalent to the many usual definitions of fibrations in the case of proper morphisms between varieties.

\begin{proposition}\label{prop:equivalent_definitions_of_fibrations}
	Let $f: X \to Y$ be a dominant morphism of finite type between excellent noetherian integral schemes.
	\begin{enumerate}
		\item \begin{itemize}[leftmargin=1em,nosep]
			      \item If $f_*\calO_X = \calO_Y$, then $f$ is a fibration;
			      \item if $f$ is a proper fibration, then $f_*\calO_X = \calO_Y$.
		      \end{itemize}
		\item \begin{itemize}[leftmargin=1em,nosep]
			      \item If $f$ is a fibration and $X$ is normal, then $\fkk(Y)$ is algebraically closed in $\fkk(X)$;
			      \item if $\fkk(Y)$ is algebraically closed in $\fkk(X)$ and $Y$ is normal, then $f$ is a fibration.
		      \end{itemize}
	\end{enumerate}
\end{proposition}
\begin{proof}
	\hspace{1em}
	\begin{step}
		Assertion (a).
	\end{step}

	If $f_*\calO_X = \calO_Y$, then by definition of relative normalization we must have $\calO_{Y'} = \calO_Y$ and hence $f$ is a fibration.
	Conversely, suppose $f$ is proper.
	Then $f_*\calO_X$ is a coherent sheaf by Finiteness of Proper Morphism (\cite[\href{https://stacks.math.columbia.edu/tag/02O5}{Tag 02O5}]{Stacks}) and hence is finite over $\calO_Y$.
	Since the integral closure of $\calO_Y$ in $f_*\calO_X$ is $\calO_Y$ itself, it enforces $\calO_Y = f_*\calO_X$.

	\begin{step}
		Assertion (b).
	\end{step}

	First we show that (fibration) $\implies$ (alg. closed) when $X$ is normal.
	Replacing $Y$ by an affine open subset and $X$ by its preimage, we may assume that $Y = \Spec A$ is affine.
	Let $\KK = \fkk(Y)$ and $\KK'$ be the algebraic closure of $\KK$ in $\fkk(X)$.
	If $\KK' \neq \KK$, then the integral closure $A'$ of $A$ in $\KK'$ is not equal to $A$.
	The inclusion $A \subset A' \subset \fkk(X)$ induces a factorization $U \xrightarrow{f'} Y' \xrightarrow{v} Y$ of $f$, where $U$ is an affine open dense subset of $X$ and $Y' = \Spec A'$, since $X \to Y$ is of finite type.
	Let $X'$ be the closure of $\Gamma_{f'} \subset U \times_Y Y'$ in $X \times_Y Y'$, where $\Gamma_{f'}$ is the graph of $f'$.
	Then we have a commutative diagram
	\[
		\begin{tikzcd}
			X' \ar[r, "u"] \ar[d,"f'"'] & X \ar[d, "f"] \ar[ld, dashed, "f'"'] \\
			Y' \ar[r, "v"] & Y.
		\end{tikzcd}
	\]
	Note that $u$ is given $X' \injmap X \times_Y Y' \to X$ and hence is finite.
	On the other hand, $u$ is birational.
	Then the normality of $X$ implies that $u_* \calO_{X'} = \calO_X$ and hence $f_* u_* \calO_{X'} = f_* \calO_X$.
	Note that the relative normalization only depends on $f'_* \calO_{X'}$.
	Hence $X' \to Y$ is also a fibration.
	By the universal property of the relative normalization, finite morphism $Y' \to Y$ factors through $Y \xrightarrow{\id} Y$, which is a contradiction since $Y' \neq Y$.

	When $Y$ is normal, for any affine open subset $U \subset Y$, $\calO_Y(U)$ is integral closed in $\fkk(Y)$ and $\fkk(Y)$ is algebraically closed in $\fkk(X)$, so $\calO_Y(U)$ is integral closed in $\fkk(X)$ and hence in $\calO_X(f^{-1}(U))$ (since $\calO_X(f^{-1}(U)) \subset \fkk(X)$).
	Then the relative normalization of $Y$ in $X$ is $Y$ itself and hence $f$ is a fibration.
\end{proof}

The following example shows that the assumption of normality in \cref{prop:equivalent_definitions_of_fibrations} are necessary.

\begin{example}[Whitney umbrella]
	Let $A = \kk[u^2]$ and
	\[ B = \kk[v,uv,u^2] = \kk[u^2] + v \kk[u] = \{f + vg \in \kk[u,v] : f \in \kk[u^2], g \in \kk[u]\}. \]
	be the coordinate ring of the Whitney umbrella (see \cref{fig:whitney_umbrella}).
	Note $B$ is not a normal ring since $v$ is integral over $B$ but not contained in $B$.
	Then $A$ is integral closed in $B$ since the integral closure of $A$ in $\Frac(B) = \kk(v,u)$ is $\kk[u]$ and $\kk[u] \cap B = A$.
	However, $\Frac(A) = \kk(u^2)$ is not algebraically closed in $\Frac(B) = \kk(v,u)$ since $u$ is algebraic over $\kk(u^2)$ but not contained in $\kk(u^2)$.

	Write $B = \kk[x,y,t]/(y^2 - x^2 t)$ with $x = v$, $y = uv$, and $t = u^2$.
	Then the fiber of $\Spec B \to \Spec A$ over the generic point of $\Spec A$ is given by $\Spec \kk(t)[x,y]/(y^2 - x^2 t)$.
	It is not geometrically irreducible when $\characteristic \kk \neq 2$ since $\kk(t)[x,y]/(y^2 - x^2 t) \ten_{\kk(t)} \kk(u) \cong \kk(u)[x,y]/(y^2 - x^2 u^2)$ has two irreducible components defined by $y = xu$ and $y = -xu$ respectively.
\end{example}

\begin{corollary}
	Let $f: X \to Y$ be a dominant morphism of finite type between excellent noetherian integral schemes.
	Suppose that $X$ is normal and separable over $Y$.
	Then the generic fiber $X_\eta$ is geometrically integral and hence a variety in our sense.
\end{corollary}
\begin{proof}
	Follows from \cref{lem:geometrically_irreducible_and_separably_close_in_function_field,lem:criterion_of_geometrical_reducedness_by_separability,prop:equivalent_definitions_of_fibrations}.
\end{proof}

The condition $f_*\calO_X = \calO_Y$ has a very clear geometric mean.

\begin{proposition}\label{lem:connected_fiber_by_pushforward_of_structure_sheaves}
	Let $f: X \to Y$ be a dominant morphism of finite type between excellent noetherian integral schemes.
	Suppose that $f_*\calO_X = \calO_Y$.
	Then $f$ has geometrically connected fibers.
\end{proposition}
\begin{proof}
	The connectedness of fiber follows from two things: one is the geometrically connectedness can be checked over all \'{e}tale covers (\cite[\href{https://stacks.math.columbia.edu/tag/03GZ}{Tag 03GZ}]{Stacks}); the other is the Theorem on Formal Function (\cite[\href{https://stacks.math.columbia.edu/tag/02OD}{Tag 02OD}]{Stacks}).

	Let $g:U \to Y$ be any \'etale morphism and $f_U:X_U \to Y_U$ be the base change of $f$ along $g$.
	By Cohomology and Flat Base Change (\cite[\href{https://stacks.math.columbia.edu/tag/02KH}{Tag 02KH}]{Stacks}), we have $(f_U)_*\calO_{X_U} = g^*(f_*\calO_X) = g^*\calO_Y=\calO_{U}$.
	Hence we only need to prove that all fiber of $f$ is connected.

	Let $y \in Y$.
	Denote by $X_{y,n} = X \times_Y \calSpec_Y \calO_Y/\frakm_y^n$.
	The Theorem on Formal Function shows that $\invlim_{n} H^0(X_{y,n},\calO_{X_{y,n}}) \cong \widehat{(f_*\calO_X)_y} = \widehat{\calO_{Y,y}}$.
	If $X_y$ has at least two irreducible components, so is $X_{y,n}$ for all $n$.
	Hence the left hand side can be written as direct sum of two rings, namely $A_1$ and $A_2$.
	Then we have $\widehat{\calO_{Y,y}} = A_1 \oplus A_2$.
	Let $e_i$ be unit in $A_i$.
	Since $\widehat{\calO_{Y,y}}$ is a local ring, $e_1e_2 = 0$ implies $e_1,e_2 \in \widehat{\frakm_{y}}$.
	However, the sum $e_1+e_2$ is a unit in $A_1 \oplus A_2$, which is a contradiction.
\end{proof}

When $f$ is proper morphism between varieties, the construction of the relative normalization is exactly the Stein factorization of $f$.

\begin{corollary}[{Stein Factorization, ref. \cite[\href{https://stacks.math.columbia.edu/tag/03H2}{Tag 03H2}]{Stacks}}]
	Let $f: X \to Y$ be a proper morphism between varieties.
	Then there exist a variety $Z$ and morphisms $g: X \to Z$ and $h: Z \to Y$ such that $f = h \circ g$, $g_* \calO_X = \calO_Z$, and $h$ is finite.
	In particular, $g$ has geometrically connected fibers.
\end{corollary}

Another application is the Zariski's main theorem.

\begin{corollary}[Zariski's Main Theorem]
	Let $f:X \to Y$ be a proper birational morphism of varieties and $Y$ is normal.
	Then $f$ has (geometrically) connected fibers.
\end{corollary}
\begin{proof}
	Follows from \cref{prop:equivalent_definitions_of_fibrations,lem:connected_fiber_by_pushforward_of_structure_sheaves}.
\end{proof}



\subsection{Valuations in general settings}

In this subsection, we review some facts about valuation on a fields.\footnote{The symbol $\nu$ is the Greek letter ``nu'' and $v$ is the English letter ``v''. Seems $v$ should be a reasonable notation of valuations, but I will also use $\nu$ sometimes.}
The references are \cite{Günther-2016-valuation-theoretic-methods-birational-geometry,Vaquie-2006-valuations}.

\begin{definition}
	A \emph{totally ordered abelian group} is an abelian group $\Gamma$ together with a total order $\geq$ satisfying $a \geq b \implies a + c \geq b + c$ for all $a,b,c \in \Gamma$.
\end{definition}

\begin{example}
	The real number $\bbR$ equipped usual addition and order is a totally ordered abelian group.
	Note that every subgroup of a totally ordered abelian group is a totally ordered abelian group.
	Hence $\bbZ,\bbQ$ with addition and usual order are also totally order abelian groups.
\end{example}

\begin{example}
	Let $\Gamma_1,\Gamma_2$ be two totally ordered abelian groups.
	Then its direct sum $\Gamma = \Gamma_1 \oplus \Gamma_2$ equipped with lexicographic order is a totally ordered abelian group.
	Indeed, if $(a_1,a_2) \geq (b_1,b_2)$, then either $a_1 > b_1$ or $a_1 = b_1, a_2 \geq b_2$.
	We have $a_1 + c_1 > b_1 + c_1$ or $a_1 + c_1 = b_1 + c_1, a_2 + c_2 \geq b_2 + c_2$ respectively.
	Hence $(a_1,a_2) + (c_1,c_2) \geq (b_1,b_2) + (c_1,c_2)$.
\end{example}

\begin{definition}
	Let $\KK$ be a field and $\Gamma$ be a totally ordered abelian group.
	A \emph{(Krull) valuation} on $\KK$ is a map $v: \KK^\times \to \Gamma$ such that for any $f, g \in \KK^\times$, we have
	\begin{enumerate}
		\item $v(fg) = v(f) + v(g)$,
		\item $v(f + g) \geq \min\{v(f), v(g)\}$.
	\end{enumerate}

	The \emph{valuation group} of $v$ is image of $v$ in $\Gamma$, denoted by $\Gamma_v$.
	Condition (a) guarantees that $\Gamma_v$ is a subgroup of $\Gamma$ and hence is also a totally ordered abelian group.

	The \emph{valuation ring} of $v$ is defined as $\calO_v = \{ f \in \KK : v(f) \geq 0 \}$, which is a local ring with maximal ideal $\frakm_v = \{ f \in \KK : v(f) > 0 \}$ and residue field $\calk_v = \calO_v / \frakm_v$.
	Two valuation $v,v'$ are said \emph{equivalent} is they define the same valuation ring.

	Let $\kk$ be a subfield of $\KK$.
	We say $v$ is a \emph{valuation of $\KK/\kk$} if $v$ is trivial on $\kk$, i.e. $v(c) = 0$ for every $c \in \kk^\times$.
	In this case, the valuation ring $\calO_v$ is a $\kk$-algebra and the residue field $\calk_v$ is an extension of $\kk$.
\end{definition}

\begin{remark}[Characterization of valuation rings]
	Let $\KK$ be a field and $A$ be a subring of $\KK$.
	Then $A$ is the valuation ring of a valuation on $\KK$ if and only if for every $f \in \KK^\times$, we have either $f \in A$ or $f^{-1} \in A$.

	The ``only if'' part is trivial.
	For the ``if'' part, we can construct a valuation as follow.
	Let $\Gamma = \KK^\times / A^\times$ be the abelian group (the operator will be written as addition)
	equipped an order given by $f>g$ if $f-g$ lies in the image of $A\setminus\{0\}$ under $\KK^\times \to \Gamma$.
	Then $\Gamma$ is a totally ordered abelian group.
	The valuation $v:\KK^\times \to \Gamma$ is the quotient map.

	For more details, see \cite[\href{https://stacks.math.columbia.edu/tag/00I8}{Tag 00I8}]{Stacks}.
\end{remark}

\begin{theorem}[{Valuations criteria for separatedness and properness, ref. \cite[\href{https://stacks.math.columbia.edu/tag/0BX4}{Tag 0BX4}]{Stacks}}]
	Let $f: X \to Y$ be a morphism of finite type between noetherian integral schemes.
	Consider the following commutative diagram:
	\[ \begin{tikzcd}
			\Spec \KK \ar[r] \ar[d] & X \ar[d, "f"] \\
			\Spec \calO_v \ar[r] \ar[ru, gray] & Y,
		\end{tikzcd} \]
	where $\KK$ is a field and $\calO_v$ is a valuation ring on $\KK$.
	Then
	\begin{enumerate}
		\item $f$ is separated if and only if for every commutative black diagram, there exists at most one gray arrow making the whole diagram commute;
		\item $f$ is proper if and only if for every commutative black diagram, there exists a unique gray morphism making the whole diagram commute.
	\end{enumerate}
	\Yang{}
\end{theorem}
\begin{proof}
	We first show the criteria for separatedness.
	Let $\Delta_{X/Y}: X \to X \times_Y X$ be the diagonal morphism.
	Then $f$ is separated if and only if $\Delta_{X/Y}$ is a closed immersion.
	If $\Delta_{X/Y}$ is a closed immersion, then for any commutative black diagram, the two morphisms $\Spec \calO_v \to X$ induced by the two gray arrows will factor through $\Delta_{X/Y}$ and hence must be equal.
	Conversely, if for any commutative black diagram there exists at most one gray arrow making the whole diagram commute, then $\Delta_{X/Y}$ is a closed immersion by the valuative criterion of properness (\cite[\href{https://stacks.math.columbia.edu/tag/0BX4}{Tag 0BX4}]{Stacks}).
	Then $f$ is separated.

	Now consider the criteria for properness.
	Let $\overline{X}$ be the closure of $X$ in some proper scheme over $Y$.
	Then $\overline{X} \to Y$ is proper and hence satisfies the valuative criterion of properness.
	Then for any commutative black diagram, there exists a unique gray arrow making the whole diagram commute.

\end{proof}

A fact is that a valuation ring is noetherian if and only if it is a DVR or a field (\cite[\href{https://stacks.math.columbia.edu/tag/00II}{Tag 00II}]{Stacks}).
On this subsection, we will focus on some more general valuation rings.

\begin{example}\label{eg:monomial_valuation}
	Let $\kk$ be a field and $\KK = \kk((x_1,\cdots,x_d))$ be the field of formal Laurent series in $d$ variable over $\kk$.
	Let $\Gamma = \bbZ^d$ equipped lexicographic order.
	We define a valuation $\nu: \KK^\times \to \Gamma$ by
	\[ f = \sum_{I \in \bbZ^d} a_I x^I \mapsto \inf \ \{I \in \bbZ^d : a_I \neq 0\}, \]
	where the infimum is taken with respect to the lexicographic order.
	Since there are only finitely many items in $f$ with negative exponents, the infimum above is indeed a minimum and hence $\nu$ is well-defined.
	By checking monomial of form $x^I$ and the fact that every $f \in \KK$ can be written as $f = a_{\nu(f)} x^{\nu(f)} + f_0$ with $\nu(f_0) > \nu(f)$, we see that $\nu$ is indeed a valuation.
	Note that when $d = 1$, then $\nu$ is just the order of a formal Laurent power series and $\calO_\nu = \kk[[x_1]]$.

	Suppose that $d > 2$.
	Then the valuation ring $\calO_\nu$ of $\nu$ is strictly large than the ring $\kk[[x_1,\cdots,x_d]]$ of formal power series since $x_1x_2^{n} \in \calO_\nu$ for all $n \in \bbZ$.
	As we have mentioned, $\calO_\nu$ is a local ring.
	Moreover, all ideals of $\calO_\nu$ forms a totally ordered set under inclusion.
	This is because $f \in \fraka$ for an ideal $\fraka \subset \calO_\nu$ will imply $g \in \fraka$ for all $g \in \KK$ with $\nu(g) \geq \nu(f)$ since $g/f \in \calO_\nu$.
	The ring $\calO_\nu$ is not noetherian since $(x_1x_2^{0}) \subset (x_1x_2^{-1}) \subset (x_1x_2^{-2}) \subset \cdots$ is a strictly ascending chains of ideals in $\calO_\nu$.

	Finally, we compute the Krull dimension of $\calO_{\nu}$.
	For simplicity, we choose the reverse lexicographic order on $\bbZ^d$, i.e. $\nu(x_d) > \nu(x_{d-1}) > \cdots > \nu(x_1)$.
	Easily see this is isomorphic to the original valuation by changing notations of indeterminates.
	Consider the chain of ideal
	\begin{equation}\label{eq:strictly_chain_of_prime_ideals_in_formal_fields}
		0 \subsetneq (x_d) \subsetneq (x_{d-1}) \subsetneq \cdots \subsetneq (x_1).
	\end{equation}
	Denote by $\calO_d$ the valuation ring $\calO_{\nu_d}$ of $\nu_d: \kk((x_1,\cdots,x_d)) \to \bbZ^d$ constructed above (with reverse lexicographic order).
	Then we have an isomorphism $\calO_d/(x_{k+1}) \to \calO_{k}$ by looking at the obvious $\kk$-basis of $\calO_d$ and $\calO_{k}$.
	Then \cref{eq:strictly_chain_of_prime_ideals_in_formal_fields} is a strictly ascending sequence of prime ideals.
	Inductively, we may assume that there is no other prime ideal can be added into $(x_d) \subsetneq (x_{d-1}) \subsetneq \cdots \subsetneq (x_1)$.
	Let $\fraka$ be an ideal with $0 \subsetneq \fraka \subsetneq (x_d)$.
	Then $x_d \not\in \fraka$ and $x_d^n \in \fraka$ for $n \gg 0$ since for any $f \in \KK$, $\nu(x_d^n) \geq \nu(f)$ for $n \gg 0$.
	Hence $\fraka$ is not prime.
	Then \cref{eq:strictly_chain_of_prime_ideals_in_formal_fields} is the strict ascending chain of prime ideals of maximal length in $\calO_{\nu_d}$.
	Thus the Krull dimension of $\calO_{\nu_d}$ is $d$.
	% \Yang{}
\end{example}

\begin{definition}
	Let $\nu: \KK^\times \to \Gamma_\nu$ be a valuation on $\KK/\kk$.
	There are three notion to measure the ``dimension'' of $\nu$:
	\begin{itemize}
		\item the \emph{dimension}: $\dim \nu \coloneqq \trdeg \calk_\nu/\kk$;
		\item the \emph{rank}: $\rank \nu \coloneqq \dim \calO_\nu$, here $\dim \calO_\nu$ is the Krull dimension of the ring $\calO_\nu$;
		\item the \emph{rational rank}: $\rank_\bbQ \nu \coloneqq \dim_\bbQ \Gamma_\nu \ten_\bbZ \bbQ$, here $\dim_\bbQ \Gamma_\nu \ten_\bbZ \bbQ$ means the dimension of $\Gamma_\nu \ten_\bbZ \bbQ$ as a $\bbQ$-vector space.
	\end{itemize}
\end{definition}

\begin{remark}
	Note that the dimension and rational rank of $v$ only depend on the residue field $\calk_v$ and valuation group $\Gamma_v$ of $v$.
	Indeed, the rank of $v$ also only depends on $\Gamma_v$.

	First note that for an ideal $\fraka \subset \calO_v$, if $a \in \fraka$ and $v(b) \geq v(a)$, then $b \in \fraka$ since $a/b \in \calO_v$.
	Hence the structure of ideals of a valuation ring can be described by its image in the valuation group.

	For any totally ordered abelian group $\Gamma$, we can define its \emph{ideal} to be the subset $I \subset \Gamma$ satisfying $\forall \gamma \in I, \gamma \geq 0$ and $\forall \gamma \in I, \forall \gamma' \in \Gamma, \gamma' \geq \gamma \implies \gamma' \in I$.
	Then there is an one-to-one corresponding between the set of ideals of $\calO_v$ and the set of ideals of $\Gamma_v$ by $\fraka \mapsto I = \{v(a)|a \in \fraka\}$.
	This corresponding is inclusion preserving.
	For an ideal $I \subset \Gamma$, we say that it is \emph{prime} if $\forall \gamma,\gamma' \geq 0 \in \Gamma, \gamma+\gamma' \in I \implies (\gamma \in I) \lor (\gamma' \in I)$.
	Then under the corresponding, prime ideal maps to prime ideal.
	Hence the Krull dimension of $\calO_v$ depends only on $\Gamma_v$.
\end{remark}

\begin{example}
	A valuation is of rank $1$ iff there is an order-preserving inclusion $\Gamma_v \subset \bbR$ of abelian groups.

	Suppose that $\Gamma_v \subset \bbR$.
	Let $0 \subsetneq \frakp \subset \calO_v$ be a non-zero prime ideal.
	Then there exists $a \in \frakp$ with $0<v(a)<\infty$.
	For any $b \neq 0 \in \frakm_v$, there exists $n$ such that $v(b^n) = nv(b) \geq v(a)$.
	Then $b^n \in \frakp$ and hence $b \in \frakp$.
	We get $\frakp = \frakm_v$ and hence $\rank v = 1$.

	Conversely, suppose that $\rank v = 1$.
	For any $\gamma,\gamma' \geq 0 \in \Gamma_v$, if $n \gamma < \gamma'$ for all $n \in \bbZ_{\geq 0}$, then the set $I = \{\delta \in \Gamma_v | n \gamma < \delta, \forall n \in \bbZ_{\geq 0} \}$ is a non-empty ideal of $\Gamma_v$.
	Further more, if $\delta,\delta' \notin I$, then $\delta \leq n\gamma$, $\delta' \leq m \gamma$ and then $\delta+\delta' \leq (n+m) \gamma$.
	Hence $I$ is a prime ideal of $\Gamma$.
	Note $\gamma \in \Gamma_{v, v> 0} \setminus I$ and thus $\rank v \geq 2$.
	Therefore, for any $\gamma,\gamma' \geq 0 \in \Gamma_v$, there exists $n \in \bbZ_{\geq 0}$ such that $n \gamma \geq \gamma'$.
	In the other word, $\Gamma_v$ satisfies the \emph{archimedean property}.

	Now we can construct the map $\varphi: \Gamma_v \to \bbR$ as follows.
	Fix $\gamma_0 > 0 \in \Gamma_v$ and define $\varphi(\gamma_0) = 1 \in \bbR$.
	For any $\gamma \in \Gamma_v$, we define
	\[ \varphi(\gamma) = \sup\ \left\{ \frac{m}{n} \;\middle|\; m,n \in \bbZ, n > 0,  m\gamma \leq n\gamma_0\right\}. \]
	By the archimedean property, the set in the definition is non-empty and upper bounded.
	Hence as a map $\varphi$ is well-defined.
	Direct check shows that $\varphi$ is an order-preserving injective homomorphism.

	The ``only if'' part was first proved by H\"older.
\end{example}

The construction of \cref{eg:monomial_valuation} can be generalized to any totally ordered abelian group as follow.

\begin{construction}
	Let $\kk$ be a field and $\Gamma$ be a totally ordered abelian group.
	Denote by $\kk^\Gamma$ the direct product of $\# \Gamma$ copies of $\kk$,
	or equivalently, the $\kk$-vector space of all set-theoretical maps from $\Gamma$ to $\kk$.
	For $a = (a_\gamma)_{\gamma \in \Gamma} \in \kk^{\Gamma}$, we define $\Supp a = \{\gamma \in \Gamma \;|\; a_\Gamma \neq 0 \}$.
	Recall that an totally order set $S$ is called \emph{well ordered} if any nonempty subset has a minimal element.
	This is equivalent to that $S$ satisfies the DCC.

	We can define the \emph{field of generalized Laurent series with exponents in $\Gamma$} as follow.
	\[ \kk((\Gamma)) \coloneqq \left\{ a \in \kk^{\Gamma} \;\middle|\; \Supp a \subset \Gamma \text{ is well ordered} \right\}. \]
	The multiplication is given by
	\[ a \cdot b = c, \quad c_\gamma = \sum_{\alpha+\beta = \gamma} a_\alpha \cdot b_\beta. \]
	Since $\Supp a, \Supp b$ are well ordered, the sum in the definition is a finite sum and hence $c_\gamma$ is well-defined.
	Moreover, $\Supp c \subset \Supp a + \Supp b$ is also well ordered and hence $c \in \kk((\Gamma))$.
	Let $x^\gamma$ be the element in $\kk^\Gamma$ given by $(x^\gamma)_\alpha = \delta_{\gamma,\alpha}$, i.e. it is equal to $1$ on the $\gamma$-component and $0$ on the others.
	Then $x^\alpha \cdot x^\beta = x^{\alpha + \beta}$.
	Formally we have
	\[ \kk((\Gamma)) = \left\{ \sum_{\gamma \in S} a_\gamma x^\gamma \;\middle|\; S \subset \Gamma \text{ well ordered} \right\}. \]

	For any $a = (a_\gamma)_{\gamma \in \Gamma} \in \kk((\Gamma))$, we can inductively construct its inverse $a^{-1} = b = (b_\gamma)_{\gamma \in \Gamma}$ as follow.
	First we assume that the minimal element of $\Supp a$ is $0$ and $a_0 = 1$, otherwise we can replace $a$ by $a_{\gamma_0}^{-1} x^{-\gamma_0} a$ for the minimal element $\gamma_0 \in \Supp a$.
	Let $S$ be the semigroup generated by $\Supp a$ in $\Gamma_{\geq 0}$.
	Then $S$ is well ordered by \cref{rmk:well_orderedness_of_sum_of_well_ordered_sets}.
	We can define
	\[ b_\gamma = \begin{dcases}
			1, & \gamma = 0,                       \\
			-\sum_{\substack{\alpha+\beta = \gamma \\ \alpha,\beta \in S, \beta < \gamma}} a_\alpha b_\beta, & \gamma > 0, \gamma \in S, \\
			0, & \gamma \not\in S.
		\end{dcases} \]
	Then for any $\gamma> 0 \in S$,
	\[ (ab)_\gamma = \sum_{\alpha+\beta = \gamma} a_\alpha b_\beta = a_0 b_\gamma + \sum_{\substack{\alpha+\beta = \gamma \\ \alpha,\beta \in S, \beta < \gamma}} a_\alpha b_\beta = 0. \]
	For $\gamma \not\in S$, we have $(ab)_\gamma = 0$ trivially.

	There is a natural valuation $v$ on $\kk((\Gamma))/\kk$ given by
	\[ v(a) \coloneqq \min \Supp a \in \Gamma. \]
	Then $\Gamma_v = \Gamma$ and $\calk_v = \kk$.
	Conversely, given a totally ordered abelian group $\Gamma$ and a field $\kk$, $\kk((\Gamma))$ seems to be the best choice of field with a valuation $v$ such that $\Gamma_v = \Gamma$ and $\calk_v = \kk$; see \Yang{}.
\end{construction}

\begin{remark}\label{rmk:well_orderedness_of_sum_of_well_ordered_sets}
	Let $\Gamma$ be a totally ordered abelian group and $A,B \subset \Gamma_{\geq 0}$ be well ordered subsets.
	Then
	\begin{enumerate}
		\item $A+B = \{a+b|a \in A, b \in B\}$ is also well ordered;
		\item the semigroup $S = \{a_1 + \cdots + a_n | n \in \bbZ_{\geq 0}, a_i \in A\}$ generated by $A$ is also well ordered.
	\end{enumerate}

	First we prove (a).
	Indeed, if $A+B$ is not well ordered, then there exists a strictly decreasing sequence $\alpha_1 > \alpha_2 > \cdots$ in $A+B$.
	Then we can write $\alpha_i = a_i + b_i$ with $a_i \in A, b_i \in B$.
	Since $A$ is well ordered, there exists a minimal element $a_n$ in the set $\{a_i|i \in \bbZ_{>0}\}$.
	Inductively, we can choose a non-decreasing subsequence $a_{i_1} = a_n, a_{i_2}, \cdots$.
	Replacing the original sequence by this subsequence, we may assume that $a_1 \leq a_2 \leq \cdots$.
	Then we have $b_1 > b_2 > \cdots$ since $\alpha_1 > \alpha_2 > \cdots$.
	This is a contradiction since $B$ is well ordered.

	Now we prove (b).
	If $\Gamma$ is archimedean, i.e., $\Gamma$ can be embedded into $\bbR$, then the result follows from (a) since if there is a strictly decreasing sequence $\alpha_1 > \alpha_2 > \cdots$ in $S$, then every $\alpha_i$ can be written as sum of $n$ elements in $A$ for some $n \in \bbZ_{\geq 0}$ satisfying $n a \geq \alpha_1$ for all $a \in A$.
	Then $\{\alpha_i\}$ is contained in the set of sums of $n$ elements in $A$, which is well ordered by (a).

	For a general totally ordered abelian group $\Gamma$,
	we can define an equivalence relation $\sim$ on $\Gamma_{\geq 0}$ by $\alpha \sim \beta$ if there exists $n \in \bbZ_{>0}$ such that $n\alpha \geq \beta$ and $n\beta \geq \alpha$.
	The order on $\Gamma$ induces an order on the quotient $\Theta = \Gamma_{\geq 0}/\sim$.
	Denote by $\pi: \Gamma_{\geq 0} \to \Theta$ the quotient map.
	Then $\pi(A) \subset \Theta$ is well ordered and $\pi(S) = \pi(A)$ since $\pi(\alpha+\beta) = \pi(\alpha)$ for any $\alpha,\beta \in \Gamma_{\geq 0}$ with $\pi(\alpha) \geq \pi(\beta)$.

	Suppose that $S$ is not well ordered, then there exists a strictly decreasing sequence $\alpha_1 > \alpha_2 > \cdots$ in $S$.
	Denote $\alpha_i^0 = \alpha_i$.
	% Since $\pi(S) = \pi(A)$ is well ordered, after truncating, we may assume that $\theta = \pi(\alpha_1) = \pi(\alpha_2) = \cdots$.
	Set $\theta = \pi(\alpha_1)$.
	Then there exists $n \in \bbZ_{>0}$ such that every $\alpha_i$ can be written as $\beta_i + \alpha_i^1$ with $\beta_i = a_{i,1} + \cdots + a_{i,n}$, $a_{i,j} \in A \cap \pi^{-1}(\theta)$ and $\alpha_i^1 \in \Gamma_{\geq 0}$ with $\pi(\alpha_i^1) < \theta$.
	Note that $\pi^{-1}(\theta)$ is archimedean and hence $\beta_i$ will be stabilized for $i \gg 0$.
	Then $\{\alpha_i^1\}_i$ will form a strictly decreasing sequence in $\Gamma_{\geq 0}$.
	Repeating the argument, we will get a strictly decreasing sequence $\pi(\alpha_1^0) > \pi(\alpha_1^1) > \pi(\alpha_1^2) > \cdots$ in $\pi(S) = \pi(A)$, which is a contradiction.
\end{remark}

Given valuation group and residue field, we can ask how a valuation can be described in terms of them.
The following theorem of Kaplansky gives an answer to this question under some specific conditions.

\begin{definition}
	Let $\Gamma$ be a totally ordered abelian group.
	A \emph{convex subgroup} of $\Gamma$ is a subgroup $C \subset \Gamma$ such that for any $x,y \in C$ and $z \in \Gamma$, if $x \leq z \leq y$, then $z \in C$.
	We say that $\Gamma$ is \emph{discrete in the sense of Kaplansky} if
	\begin{itemize}
		\item the set of convex subgroups of $\Gamma$ is well-ordered by inclusion;
		\item for any two consecutive convex subgroups $C_1 \subsetneq C_2$ of $\Gamma$, the quotient $C_2/C_1$ is isomorphic to $\bbZ$ as an ordered abelian group.
	\end{itemize}
\end{definition}

\begin{example}
	The totally ordered abelian group $\bbZ^d$ with lexicographic order is discrete in the sense of Kaplansky.
	Indeed, the convex subgroups of $\bbZ^d$ are $0 \subsetneq \bbZ e_1 \subsetneq \bbZ e_1 + \bbZ e_2 \subsetneq \cdots \subsetneq \bbZ e_1 + \cdots + \bbZ e_d = \bbZ^d$.
\end{example}

\begin{example}
	Let $\Gamma = \{a + b \sqrt{2} | a,b \in \bbZ\} \subset \bbR$ with the natural order.
	Then $\Gamma$ is a totally ordered abelian group and the only convex subgroups of $\Gamma$ are $0$ and $\Gamma$.
	Hence $\Gamma$ is not discrete in the sense of Kaplansky since $\Gamma/0 \cong \bbZ + \bbZ\sqrt{2}$ is not isomorphic to $\bbZ$ as an ordered abelian group.
\end{example}

	\[ \Div(\frakX) \coloneqq \CaDiv(\frakX) \coloneqq \dirlim_{Y \in \frakX} \Div(Y). \]
	Similar definitions for $\Pic(\frakX)$, $\NS(\frakX)$, $\upN^1(\frakX)$, etc.
	Elements of $\Div(\frakX)$ (resp. $\Pic(\frakX)$, $\NS(\frakX)$, $\upN^1(\frakX)$) are called \emph{$\bfb$-Cartier divisors} (resp. \emph{linearly equivalent classes of $\bfb$-Cartier divisors}, \emph{numerical equivalence classes of $\bfb$-Cartier divisors}, \emph{real numerical equivalence classes of $\bfb$-Cartier divisors}).

	For cycles, we define
	\[ Z_{k}(\frakX) \coloneqq \invlim_{Y \in \frakX} Z_{k}(Y), \]
	Similar definitions for $\CH_{k}(\frakX)$, $\upN_k(\frakX)$, etc.
	Elements of $Z_{k}(\frakX)$ (resp. $\CH_{k}(\frakX)$, $\upN_1(\frakX)$) are called \emph{$\bfb$-cycles} (resp. \emph{$\bfb$-cycle classes}, \emph{numerical equivalence classes of $\bfb$-cycles}).
	If we only consider normal birational models, we will also use the notation $\WDiv(\frakX) \coloneqq Z_{d-1}(\frakX_\nor)$, $\WCl(\frakX) \coloneqq \CH_{d-1}(\frakX_\nor)$.
	%  \Yang{}
\end{definition}

For the cohomological objects, for instance, $\Div(\frakX)$, we have the following description:
an element of $\Div(\frakX)$ is represented by a divisor $D_Y$ on some birational model $Y$ of $X$, and two divisors $D_Y$ and $D_{Y'}$ represent the same element in $\Div(\frakX)$ if there exists a birational model $Z$ dominating both $Y$ and $Y'$ such that the pullbacks of $D_Y$ and $D_{Y'}$ to $Z$ are equal.

For the homological objects, for instance, $\WDiv(\frakX)$, we have the following description:
an element of $\WDiv(\frakX)$ is represented by a collection of divisors $D_Y$ on all birational models $Y$ of $X$ such that for any birational morphism $f: Y' \to Y$, we have $D_{Y} = f_* D_{Y'}$.
