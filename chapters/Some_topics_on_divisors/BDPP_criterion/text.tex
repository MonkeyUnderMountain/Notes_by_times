\section{BDPP's criterion}


\subsection{Movable curves}


  \begin{definition}
    A curve $C$ on a projective variety $X$ is called \emph{movable} if it belongs to an algebraic family of curves that cover $X$, i.e. there exist a variety $Y$ and a family of curves $\{C_y\}_{y \in Y}$ such that $C = C_{y_0}$ for some $y_0 \in Y$ and $\bigcup_{y \in Y} C_y = X$.
    \Yang{}
  \end{definition}


  \begin{definition}
    A curve $C$ is called \emph{strongly movable} if there exist a birational morphism $\pi: \tilde{X} \to X$ from a normal projective variety $\tilde{X}$ and a family of curves $\{\tilde{C}_y\}_{y \in Y}$ that cover $\tilde{X}$ such that $C = \pi_* \tilde{C}_{y_0}$ for some $y_0 \in Y$.
    \Yang{}
  \end{definition}


  \begin{proposition}\label{prop:inclusions_of_movable_and_strongly_movable_cone_of_curves}
    We have inclusions
    \[ \upoperatorname{SMov}_1(X) \subseteq \Mov_1(X) \subseteq \Nef_1(X) \]
    on $\upN_1(X)$.
    \Yang{}
  \end{proposition}



\subsection{BDPP's criterion}

  Boucksom, Demailly, P\u{a}un and Peternell proved that the inclusions in \cref{prop:inclusions_of_movable_and_strongly_movable_cone_of_curves} are actually equalities.

  \begin{theorem}[{BDPP's criterion, ref. \cite[Theorem 0.2]{BDPP12}}]
    Let $X$ be a projective variety over $\kkk$.
    Then a numerical Cartier divisor class $D$ is pseudo-effective if and only if $D \cdot C \geq 0$ for every strongly movable curve $C$ on $X$.
    \Yang{}
  \end{theorem}
  \begin{proof}
    \Yang{}
  \end{proof}

  \begin{remark}
    Although we only prove the characteristic $0$ case, the theorem is actually true in any characteristic.
    For example, see arxiv:1711.10901.
    \Yang{}
  \end{remark}

  \begin{remark}
    We can also define the cone $\Nci_1(X)$ of complete intersection of nef divisors, which is the closure of the cone generated by classes of the form $A_1 \cdots A_{d-1}$ where $A_i$ are nef divisors on $X$.
    Then in general we have $\Nci_1(X) \subseteq \Mov_1(X)$, and the inclusion can be strict.

    \Yang{}
  \end{remark}

  \begin{example}
    Let $X$ be a smooth projective surface and $C$ be a curve on $X$.
    Then $C$ is movable if and only if $C^2 \geq 0$ and $C \cdot A > 0$ for some ample divisor $A$ on $X$.
     \Yang{}
  \end{example}
