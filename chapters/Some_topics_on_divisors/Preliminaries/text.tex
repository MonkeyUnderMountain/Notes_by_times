\section{Preliminaries}

  Let $\kkk$ be an algebraically closed field, usually of characteristic $0$.
  Let $X$ be a normal projective variety over $\kkk$ of dimension $d$.


\subsection{Resolution of singularities}

  \begin{theorem}
    Let $X$ be a normal projective variety over $\kkk$.
    Then there exists a smooth projective variety $Y$ and a birational morphism $f: Y \to X$.
  \end{theorem}
