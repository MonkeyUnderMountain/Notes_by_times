\begin{theorem}
  We can define a global period map
  \[ \bbP^1 \setminus \{0,1,\infty\} \to \Gamma_2 \backslash \calH \to \SL_2(\bbZ) \backslash \calH, \quad \lambda \mapsto \frac{B(\lambda)}{A(\lambda)}. \]
  Here 
  \[ A(\lambda) = \int_{\delta} \frac{d x}{\sqrt{x(x-1)(x-\lambda)}}, \quad B(\lambda) = \int_{\gamma} \frac{d x}{\sqrt{x(x-1)(x-\lambda)}}, \]
  and $\delta,\gamma$ be the generators of $H^1(X,\bbZ)$, $X = V(y^2-x(x-1)(x-\lambda)) \subset \bbP^2(\bbC)$.
\end{theorem}

\begin{remark}
  The map $\bbP^1 \setminus \{0,1,\infty\} \to \Gamma_2 \backslash \calH$ is isomorphic;
  $ \Gamma_2 \backslash \calH \to \SL_2(\bbZ) \backslash \calH$ is $6:1$.
  Note $\SL_2(\bbZ) \backslash \calH$ parameter elliptic curves.
  Hence for a general elliptic curve $E$, there exists $6$ different $\lambda$ such that $E \cong V(y^2-x(x-1)(x-\lambda)) \subset \bbP^2(\bbC)$.
\end{remark}

\begin{theorem}
  The period map gives
  \begin{enumerate}
    \item $\bbD \to \calH$ between universal covering space is biholomorphic;
    \item The global monodromy representation $\pi_1(\bbP^1 \setminus \{0,1,\infty\}) \to \SL_2(\bbZ)$ is injective.
  \end{enumerate}
\end{theorem}
