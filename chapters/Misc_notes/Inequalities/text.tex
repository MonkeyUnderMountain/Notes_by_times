\section{Estimation and Inequalities}

  In this section, fix a field $\kk$, $X$ a projective variety of dimension $d$ over $\kk$, $L$ an ample line bundle on $X$ and $f : X \dashrightarrow X$ a dominant rational self-map.
  Set $\lambda_i = \lambda_i(f)$ and $\mu_i = \mu_i(f)$ for every $i = 0, 1, \ldots, d$.
  Set $\mu_{d+1} = 0$ for convenience.


\subsection{Sequence with exponential growth}

  In this subsection, we will introduce some elementary notations and lemmas about sequences with exponential growth.

  \begin{definition}
    We say that a sequence of positive numbers $a_n$ has \emph{exponential growth} if $\lim_{n \to \infty} a_n^{1/n}$ exists and is positive.
  \end{definition}

  \begin{definition}
    Let $a_n$ and $b_n$ be two sequences of exponential growth.
    We write $a_n \preccurlyeq_e b_n$ (resp. $a_n \prec_e b_n$) if $\limsup_{n \to \infty} a_n/b_n \leq 1$ (resp. $<1$), and write $a_n \asymp_e b_n$ if $a_n \preccurlyeq_e b_n$ and $a_n \succcurlyeq_e b_n$.
  \end{definition}


  \begin{lemma}\label{lemma:elementary_inequalities}
    Suppose that $a_n \asymp_e a^n, b_n \asymp_e b^n$, then
    \begin{enumerate}
      \item $a_n b_n \asymp_e (ab)^n$;
      \item $a_n/b_n \asymp_e (a/b)^n$;
      \item if $a_n \succ_e b_n$, then $a_n > b_n$ for every sufficiently large $n$;
      \item if $a \geq b$, then $a_n + b_n \asymp_e a^n$;
      \item if $a_n \geq b_n$ for every $n$ and $a > b$, then $a_n - b_n \asymp_e a^n$.
    \end{enumerate}
  \end{lemma}

  \begin{lemma}\label{lemma:recursive_inequality_implies_exponential_growth}
    Let $x_n$ be a sequence of positive numbers and $a>b>0$ be some numbers.
    Suppose that 
    \[ x_{n+2} + ab x_n \geq (a+b) x_{n+1}, \quad \forall n \geq 0 \]
    and $x_1 > b x_0$.
    Then we have $x_n \succcurlyeq_e a^n$.
  \end{lemma}
  \begin{proof}
    The inequality can be reformulated as
    \[ x_{n+2} - b x_{n+1} \geq a (x_{n+1} - b x_n). \]
    Set $y_n := x_{n+1} - b x_n$.
    Then we have $y_{n+1} \geq a y_n$ for every $n \geq 0$ and $y_0 = x_1 - b x_0 > 0$.
    Hence, $y_n \geq a^n y_0 > 0$ for every $n \geq 0$.
    This implies $x_{n+1} = y_n + b x_n \geq y_n \succcurlyeq_e a^n$.
  \end{proof}


\subsection{Inequalities}


  Set 
  \[ L_m := (f^m)^* L, \quad M_m(t) := L_{2m} + t^m L. \]
  Let $Z$ be a subvariety of $X$ (or more generally, a cycle in $\CH^\bullet(X)$),  set 
  \[ \Sigma(t;Z)_m := \frac{1}{\dim Z} \frac{\left(M(t)_m|_Z\right)^{\dim Z}}{\left(M(t)_m|_Z\right)^{\dim Z - 1} \cdot L_m|_Z} = \frac{1}{\dim Z} \frac{M(t)_m^{\dim Z} \cdot [Z]}{M(t)_m^{\dim Z - 1} \cdot L_m \cdot [Z]}. \]
  When $Z = X$, we simply write $\Sigma(t)_m$ for $\Sigma(t;X)_m$.
  By Siu's criterion (\cref{thm:generalized_Siu_inequality}), we have $\bigl(M(t)_m - a L_m\bigr)|_Z$ is big on $Z$ provided $\Sigma(t)_m > a$.

  On the denominator of $\Sigma(t)_m$, there is some item of the form $L_{2m}^i \cdot L_m \cdot L^{d-1-i}$.
  This reminds us to estimate the intersection numbers of the form 
  \[ L_{m_1}^{r_1} \cdot L_{m_2}^{r_2} \cdots L_{m_s}^{r_s}, \]
  where $m_1 \geq m_2 \geq \cdots \geq m_s$ and $r_1 + r_2 + \cdots + r_s = d$.
  In \cite{Xie24recursive_inequalities}, such intersection numbers are called \emph{mixed degrees}.

  \begin{proposition}\label{prop:estimation_of_mixed_degrees}
    Let $r_1 + \cdots + r_s = d$ be a partition of $d$ into positive integers, and $R_i := r_1 + \cdots + r_i$.
    Then for every $\bfm = (m_1, \ldots, m_s) \in \bbZ_{\geq 0}^s$ with $m_1 \geq m_2 \geq \cdots \geq m_s$, we have
    \[ E(\bfm) \cdot \prod_i \deg_{R_i}(f^{\Delta m_i}) \leq (L_{m_1}^{r_1} \cdot L_{m_2}^{r_2} \cdots L_{m_s}^{r_s}) \leq C_{\mix} \cdot \prod_i \deg_{R_i}(f^{\Delta m_i}), \]
    where $\Delta m_i := m_i - m_{i+1}$ with $m_{s+1} := 0$ and
    \begin{itemize}
      \item $C_{\mix}$ is some constant depending only on $(X,L)$;
      \item $E(\bfm)$ is some error term satisfying 
      \[ E(\bfm) \asymp_e 1 \quad \text{ if } \bfm = (k_1m, \ldots, k_sm) \text{ for some } k_i \in \bbZ_{>0} \text{ and } m \to \infty. \]
    \end{itemize}
    In particular, if $\bfm = (k_1m, \ldots, k_dm)$ for some $k_1 \geq k_2 \geq \cdots \geq k_d \in \bbZ_{>0}$, then we have
    \[ (L_{m_1} \cdot L_{m_2} \cdots L_{m_s}) \asymp_e \prod_i \lambda_{i}^{\Delta m_i} = \mu_1^{m_1} \cdot \mu_2^{m_2} \cdots \mu_d^{m_d} \quad \text{ as } m \to \infty. \]
  \end{proposition}
  \begin{proof}
    \hspace{1em}
    \begin{step}\label{step1_in_proof_of_estimation_of_mixed_degrees}
      $(L_{m_1}^{r_1} \cdot L_{m_2}^{r_2} \cdots L_{m_s}^{r_s}) \leq C_{\mix} \prod_i \deg_{R_i}(f^{\Delta m_i})$.
    \end{step}
    By Siu's inequality, we have
    \begin{align*}
      (L_{m_2}^{d}) (L_{m_1}^{r_1} \cdot L_{m_2}^{r_2} \cdots L_{m_s}^{r_s}) &\leq \binom{d}{r_1} (L_{m_1}^{r_1} \cdot L_{m_2}^{d-r_1}) (L_{m_2}^{R_2} \cdots L_{m_s}^{r_s}).
    \end{align*}
    By the projection formula, we have 
    \begin{align*}
      L_{m_2}^d &= \deg(f^{m_2}) L^d, \\
      L_{m_1}^{r_1} \cdot L_{m_2}^{d-r_1} &= \deg(f^{m_1}) L^{r_1}_{m_1-m_2} \cdot L^{d-r_1} = \deg(f^{m_2}) \deg_{r_1}(f^{m_1 - m_2}) = \deg(f^{m_2}) \deg_{R_1}(f^{\Delta m_1}).
    \end{align*}
    Hence 
    \[ (L_{m_1}^{r_1} \cdot L_{m_2}^{r_2} \cdots L_{m_s}^{r_s}) \leq \frac{1}{L^d} \binom{d}{r_1} \deg_{R_1}(f^{\Delta m_1}) (L_{m_2}^{R_2} \cdots L_{m_s}^{r_s}). \]
    Inducting on $s$, we can find a constant $C$ depending only on $(X,L)$ and $r_i$'s such that
    \[ (L_{m_1}^{r_1} \cdot L_{m_2}^{r_2} \cdots L_{m_s}^{r_s}) \leq C \prod_i \deg_{R_i}(f^{\Delta m_i}). \]
    Since there are only finitely many partitions of $d$, we can find a uniform constant $C_{\mix}$ depending only on $(X,L)$.


    \begin{step}
      $ E(\bfm) \prod_i \deg_{R_i}(f^{\Delta m_i}) \leq (L_{m_1}^{r_1} \cdot L_{m_2}^{r_2} \cdots L_{m_s}^{r_s})$.
    \end{step}
    By Siu's inequality, we have
    \[ (L_{m_{2}}^{r_{1}+r_2} \cdot \eta) (L_{m_1}^{r_1} L_{m_{3}}^{r_{2}} \cdot \eta) \leq \binom{R_2}{r_{1}} (L_{m_1}^{r_1} L_{m_{2}}^{r_{2}} \cdot \eta) (L_{m_{2}}^{r_{1}} L_{m_{3}}^{r_{2}} \cdot \eta), \quad \eta = L_{m_3}^{r_3} \cdots L_{m_s}^{r_s}. \]
    That is,
    \begin{align*}
      (L_{m_1}^{r_1} \cdots L_{m_s}^{r_s}) & \geq \frac{1}{C_{R_2}^{r_1}} \frac{(L_{m_2}^{r_1+r_2} L_{m_3}^{r_3} \cdots L_{m_s}^{r_s}) (L_{m_1}^{r_1} L_{m_{3}}^{r_{2} + r_3} \cdots L_{m_s}^{r_s})}{(L_{m_2}^{r_1} L_{m_{3}}^{r_{2} + r_3} \cdots L_{m_s}^{r_s})}.
    \end{align*} 
    By \cref{step1_in_proof_of_estimation_of_mixed_degrees} and inducting on $s$, we may assume that 
    \begin{align*}
      (L_{m_2}^{r_1+r_2} L_{m_3}^{r_3} \cdots L_{m_s}^{r_s}) & \geq E_1(\bfm) \deg_{R_2}(f^{\Delta m_2}) \cdot \prod_{i=3}^{s} \deg_{R_i}(f^{\Delta m_i}), \\
      (L_{m_1}^{r_1} L_{m_{3}}^{r_{2} + r_3} \cdots L_{m_s}^{r_s}) & \geq E_2(\bfm) \deg_{R_1}(f^{\Delta m_1 + \Delta m_2}) \cdot \prod_{i=3}^{s} \deg_{R_i}(f^{\Delta m_i}), \\
      (L_{m_2}^{r_1} L_{m_{3}}^{r_{2} + r_3} \cdots L_{m_s}^{r_s}) & \leq C_3 \deg_{R_1}(f^{\Delta m_2}) \cdot \prod_{i=3}^{s} \deg_{R_i}(f^{\Delta m_i}).
    \end{align*}
    Combining the above inequalities, we get
    \begin{align*}
      (L_{m_1}^{r_1} \cdots L_{m_s}^{r_s}) 
      & \geq 
      \frac{E_1(\bfm) E_2(\bfm)}{C_3 C_{R_2}^{r_1}}
      \cdot \frac{\deg_{R_{1}}(f^{\Delta m_{1} + \Delta m_{2}})}{\deg_{R_{1}}(f^{\Delta m_{1}}) \cdot \deg_{R_{1}}(f^{\Delta m_{2}})}
      \cdot \prod_{i=1}^{s} \deg_{R_i}(f^{\Delta m_i}) 
    \end{align*}
    Set 
    \[ E(\bfm) := \frac{E_1(\bfm) E_2(\bfm)}{C_3 C_{R_2}^{r_1}} \cdot \frac{\deg_{R_{1}}(f^{\Delta m_{1} + \Delta m_{2}})}{\deg_{R_{1}}(f^{\Delta m_{1}}) \cdot \deg_{R_{1}}(f^{\Delta m_{2}})}. \]
    Easily to check that $E(\bfm)$ satisfies the desired property.
  \end{proof}

  \begin{lemma}\label{lemma:estimation_of_Sigma}
    We have $\Sigma(\mu_i\mu_{i+1})_m \asymp_e \mu_i^m$ for every $i = 0, 1, \ldots, d$.
  \end{lemma}
  \begin{proof}
    We have 
    \begin{align*}
      M(\mu_i \mu_{i+1})_m^d & = \sum_{j=0}^{d} \binom{d}{j} L_{2m}^{j} \cdot (\mu_i^m \mu_{i+1}^m L)^{d-j} \\
      & \asymp_e \sum_{j=0}^{d} \mu_1^{2m} \cdots \mu_j^{2m} \cdot \mu_i^{m(d-j)} \mu_{i+1}^{m(d-j)} \\
      & \asymp_e \mu_1^{2m} \cdots \mu_i^{2m} \cdot \mu_{i}^{m(d-i)} \mu_{i+1}^{m(d-i)}
    \end{align*}
    and 
    \begin{align*}
      M(\mu_i \mu_{i+1})_m^{d-1} \cdot L_m & = \sum_{j=0}^{d-1} \binom{d-1}{j} \left(L_{2m}^{j} \cdot L_m \cdot L^{d-1-j} \right) \cdot (\mu_i^m \mu_{i+1}^m)^{d-j-1}\\
      & \asymp_e \sum_{j=0}^{d-1} \mu_1^{2m} \cdots \mu_j^{2m} \cdot \mu_{j+1}^m \cdot \mu_i^{m(d-1-j)} \mu_{i+1}^{m(d-j-1)} \\
      & \asymp_e \mu_1^{2m} \cdots \mu_{i}^{2m} \cdot \mu_{i+1}^m \cdot \mu_i^{m(d-1-i)} \mu_{i+1}^{m(d-1-i)}.
    \end{align*}
    Hence
    \begin{align*}
      \Sigma(\mu_i\mu_{i+1})_m & = \frac{(L_{2m} + \mu_i^m \mu_{i+1}^m L)^{d}}{(L_{2m} + \mu_i^m \mu_{i+1}^m L)^{d-1} \cdot L_m} \\
      & \asymp_e \frac{\mu_1^{2m} \cdots \mu_i^{2m} \cdot \mu_{i}^{m(d-i)} \mu_{i+1}^{m(d-i)}}{\mu_1^{2m} \cdots \mu_{i}^{2m} \cdot \mu_{i+1}^m \cdot \mu_i^{m(d-1-i)} \mu_{i+1}^{m(d-1-i)}} \\
      & \asymp_e \mu_i^m.
    \end{align*}
    We are done.
  \end{proof}

  \begin{theorem}\label{thm:big_inequality_by_mu_i}
    For every $i = 0, 1, \ldots, d$ and $\varepsilon \in (0, 1)$, there exists $m_\varepsilon > 0$ such that for all $m \geq m_\varepsilon$,
    \[  L_{2m} - \varepsilon^m \mu_i^m L_m + \mu_i^m \mu_{i+1}^m L \]
    is big.
  \end{theorem}
  \begin{proof}
    For every $\varepsilon \in (0, 1)$, there exists $m_\varepsilon > 0$ such that for every $m \geq m_\varepsilon$, we have $\Sigma(\mu_i \mu_{i+1})_m > \varepsilon^m \mu_i^m$.
    By Siu's criterion, this implies that $M(\mu_i \mu_{i+1})_m - \varepsilon^m \mu_i^m L_m$ is big.
    % That is, $L_{2m} - \varepsilon^m \mu_i^m L_m + \mu_i^m \mu_{i+1}^m L$ is big for sufficiently large $m$.
  \end{proof}



\subsection{Estimation of dynamical degrees}

  In this subsection, we will try to estimate $\lambda_i$ using only finitely many mixed degrees.
  % The key observation is the following elementary lemma about recursive inequalities.


  Given numbers $\alpha_1 \geq \ldots \geq \alpha_{d} \in \bbR_{>0}$, and $\delta, \varepsilon \in (0, 1)$.
  Set $\beta_i = \alpha_1 \cdots \alpha_i$.
  We use $\frakA = (\alpha_1, \ldots, \alpha_d; \delta, \varepsilon)$ to denote the above data.
  Set 
  \[ \Theta(\frakA; i)_m := C_{\mix} \cdot \delta^m \cdot \sum_{j=0}^{i-1} \frac{\deg_{i-j-1}(f^{m})}{\varepsilon^{(i-j)m}\beta_{i-j-1}^m} \cdot \alpha_i^m \deg_{i-1}(f^{m}), \]
  where $C_\mix$ is the constant in \cref{prop:estimation_of_mixed_degrees}.
  We will see this number appears naturally in the estimation of $\lambda_i$ (\cref{eq:inequality_in_estimation_of_dynamical_degrees_define_Theta}).
  When choosing $\frakA = (\mu_1, \ldots, \mu_d; \delta, \varepsilon)$, we have 
  \[ \Theta(\frakA; i)_m \asymp_e \left( \varepsilon^{-i} \delta \lambda_i \right)^m. \]

  \begin{definition}
    We define the following conditions for $i = 1, \ldots, d$: 
    \begin{itemize}
      \item condition $I(\frakA,i,m)$: $\Sigma(\alpha_i \alpha_{i-j} \delta; L_m^{j})_{m} > \varepsilon^m \alpha_{i-j}^m$ for every $j = 0, 1, \ldots, i-1$;
      \item condition $J(\frakA,i,m)$: $\beta_i^m \varepsilon^{mi} \geq \varepsilon^{-mi} \Theta(\frakA; i)_m + \beta_i^m \varepsilon^{2mi}$;
      \item condition $K(\frakA,i,m)$: $\deg_i(f^{2m}) - \varepsilon^{-mi}\Theta(\frakA; i)_m \cdot \deg_i(f^{m}) > 0$.
    \end{itemize}
  \end{definition}

  Note that all of the above conditions depend only on the intersection numbers of $L_{2m}, L_m, L$.


  \begin{theorem}\label{thm:conditions_IJK_implies_estimation_of_dynamical_degrees}
    Given a data $\frakA = (\alpha_1, \ldots, \alpha_d; \delta, \varepsilon)$.
    If conditions $I(\frakA,i,m)$, $J(\frakA,i,m)$ and $K(\frakA,i,m)$ hold, then $\lambda_i \geq \varepsilon^{2i} \beta_i$.
    % \Yang{}
  \end{theorem}
  \begin{proof}
    \hspace{1em}
    \begin{step}
      Using condition $I(\frakA,i,m)$ to get \cref{eq:inequality_from_condition_I_for_infinite_n_in_estimation_of_dynamical_degrees} for all $n \geq 0$.
    \end{step}

    % There exists a normal birational model $X'$ of $X$ such that $L_m$ is ample on $X'$.
    We may suppose that $X$ is normal and $L_m$ is ample on $X$.
    Fix $n \geq 0$.
    There exists another normal birational model $X'$ of $X$ such that $f^{nm}$ is a morphism $X' \to X$.
    By generic flatness, there exists a non-empty open subset $U$ of $X$ such that $f^{nm}$ is flat over $U$.
    By Bertini's theorem, there is a normal subvariety $Z$ of $X$ given by cutting by $j$ general members of $|L_m|$ intersecting $U$.
    Denote by $W_1, \ldots, W_r$ the irreducible components of $(f^{nm})^{-1}(Z)$.
    Since $f^{nm}$ is flat at the generic point of $Z$, we have $(f^{nm})^*[Z] = \sum_i a_i [W_i]$ as cycles for some positive integers $a_i$'s.

    By Siu's criterion and condition $I(\frakA,i,m)$, the divisor $(L_{2m} + \alpha_i^m \alpha_{i-j}^m \delta^m L - \varepsilon^m \alpha_{i-j}^m L_m)|_{Z}$ is big on $Z$ for every $j = 0, 1, \ldots, i-1$.
    Then 
    \[ (f^{nm}_t)^*\bigl((L_{2m} + \alpha_i^m \alpha_{i-j}^m \delta^m L - \varepsilon^m \alpha_{i-j}^m L_m)|_{Z}\bigr) \] 
    is big on $W_t$ for every $j = 0, 1, \ldots, i-1$ and $t = 1, \ldots, r$.
    Suppose that it is represented by an effective $\bbR$-divisor $D_{t,j}$ on $W_t$.
    Hence 
    \[ (f^{nm})^*\bigl((L_{2m} + \alpha_i^m \alpha_{i-j}^m \delta^m L - \varepsilon^m \alpha_{i-j}^m L_m) \cdot L_m^{j} \bigr) = \sum_t a_t (\iota_t)_*D_{t,j} \in \CH^{j+1}(X')_\bbR \]
    is represented by an effective cycle, where $\iota_t : W_t \to X'$ is the natural inclusion.
    Intersecting with $L_{(n+2)m}^{i-j-1} \cdot L^{d-i}$, we get
    \begin{equation}\label{eq:inequality_from_condition_I_for_infinite_n_in_estimation_of_dynamical_degrees}
      L_{(n+2)m}^{i-j} \cdot L_{(n+1)m}^j \cdot L^{d-1} + \alpha_{i}^m \alpha_{i-j}^m \delta^m L_{(n+2)m}^{i-j-1} \cdot L_{(n+1)m}^{j} \cdot L_{nm} \cdot L^{d-i} \geq \varepsilon^m \alpha_{i-j}^m L_{(n+2)m}^{i-j-1} \cdot L_{(n+1)m}^{j+1} \cdot L^{d-i}.
    \end{equation}


    \begin{step}
      Show \cref{eq:main_inequality_in_step2_in_estimation_of_dynamical_degrees} of $\deg_i f^{nm}$.
    \end{step}

    Divide \cref{eq:inequality_from_condition_I_for_infinite_n_in_estimation_of_dynamical_degrees} by $\varepsilon^{(i-j)m} \beta_{i-j}^m$, we get
    \[
      \frac{L_{(n+2)m}^{i-j} \cdot L_{(n+1)m}^j \cdot L^{d-1}}{\varepsilon^{(i-j)m} \beta_{i-j}^m} + \frac{\alpha_{i}^m \delta^m L_{(n+2)m}^{i-j-1} \cdot L_{(n+1)m}^{j} \cdot L_{nm} \cdot L^{d-i}}{\varepsilon^{(i-j)m} \beta_{i-j-1}^m} \geq \frac{L_{(n+2)m}^{i-j-1} \cdot L_{(n+1)m}^{j+1} \cdot L^{d-i}}{\varepsilon^{(i-j-1)m} \beta_{i-j-1}^m}.
    \]
    By \cref{prop:estimation_of_mixed_degrees}, we have
    \[ L_{(n+2)m}^{i-j-1} \cdot L_{(n+1)m}^{j} \cdot L_{nm} \cdot L^{d-i} \leq C_{\mix} \cdot \deg_{i-j-1}(f^{m}) \cdot \deg_{i-1}(f^{m}) \cdot \deg_i(f^{nm}). \]
    Then the above inequality can be rewritten as
    \begin{equation}\label{eq:inequality_in_estimation_of_dynamical_degrees_define_Theta}
      \begin{aligned}
        \frac{L_{(n+2)m}^{i-j} \cdot L_{(n+1)m}^j \cdot L^{d-1}}{\varepsilon^{(i-j)m} \beta_{i-j}^m} - &\frac{L_{(n+2)m}^{i-j-1} \cdot L_{(n+1)m}^{j+1} \cdot L^{d-i}}{\varepsilon^{(i-j-1)m} \beta_{i-j-1}^m} \\
        & \geq - \left( C_{\mix} \cdot \delta^m \cdot \frac{\deg_{i-j-1}(f^{m}) }{\varepsilon^{(i-j)m} \beta_{i-j-1}^m} \cdot \alpha_i^m \deg_{i-1}(f^{m}) \right) \cdot \deg_i(f^{nm}).
      \end{aligned}
    \end{equation}

    Summing over $j = 0, 1, \ldots, i-1$, we get
    \[ \frac{L_{(n+2)m}^i \cdot L^{d-i}}{\varepsilon^{im} \beta_i^m} -  (L_{(n+1)m}^{i} \cdot L^{d-i}) \geq - \Theta(\frakA; i)_m \cdot \deg_i(f^{nm}). \]
    That is,
    \begin{equation}\label{eq:main_inequality_in_step2_in_estimation_of_dynamical_degrees}
      \deg_i f^{(n+2)m} - \varepsilon^{im} \beta_i^m \deg_i f^{(n+1)m} + \varepsilon^{im} \beta_i^m \Theta(\frakA; i)_m \cdot \deg_i(f^{nm}) \geq 0. 
    \end{equation}
    

    \begin{step}
      Finish the proof by \cref{lemma:recursive_inequality_implies_exponential_growth}.
    \end{step}

    The condition $J(\frakA,i,m)$ implies that
    \[ \alert{\deg_i f^{(n+2)m}} + \left(\varepsilon^{2im} \beta_i^m \cdot \varepsilon^{-mi} \Theta(\frakA; i)_m\right) \cdot \alert{\deg_i(f^{nm})} \geq (\varepsilon^{2im} \beta_i^m + \varepsilon^{-mi} \Theta(\frakA; i)_m) \cdot \alert{\deg_i f^{(n+1)m}}. \]
    While the condition $K(\frakA,i,m)$ gives the initial condition $\deg_i f^{2m} > \varepsilon^{-mi} \Theta(\frakA; i)_m \cdot \deg_i(f^{m})$.
    Then by \cref{lemma:recursive_inequality_implies_exponential_growth}, we have $\deg_i f^{nm} \succcurlyeq_e (\varepsilon^{2im} \beta_i^m)^n$.
    Hence $\lambda_i \geq \varepsilon^{2i} \beta_i$.
  \end{proof}

  
  \begin{theorem}\label{thm:lambda_i_satisfy_conditions_IJK}
    Suppose that $\mu_i > \mu_{i+1}$.
    For every $\varepsilon_0 \in (0, 1)$, there exists $\delta \in (0, 1)$ and $\varepsilon \in (\varepsilon_0, 1)$ such that for every sufficiently large $m$, conditions $I(\frakA,i,m)$, $J(\frakA,i,m)$ and $K(\frakA,i,m)$ hold for $\frakA = (\mu_1, \ldots, \mu_d; \delta, \varepsilon)$.
  \end{theorem}
  \begin{proof}
    Choose $\delta \in (0, 1)$ such that $\delta \mu_i > \mu_{i+1}$ and $\varepsilon \in (\varepsilon_0, 1) \cap (\delta^{1/3d}, 1)$.

    By \cref{lem:growth_of_sigma_L_m}, there exists $m_1 > 0$ such that for every $m \geq m_1$, we have $\Sigma(\mu_i \mu_{i-j} \delta; L_m^j)_m > \varepsilon^m \mu_{i-j}^m$ for every $j = 0, 1, \ldots, i-1$.
    Hence condition $I(\frakA,i,m)$ holds for every $m \geq m_1$.

    For condition $J(\frakA,i,m)$, we have
    \begin{align*}
      \varepsilon^{-mi}\Theta(\frakA; i)_m + \lambda_i^m \varepsilon^{2mi} & \asymp_e \varepsilon^{-2mi} \cdot \delta^m \cdot \lambda_i^m + \varepsilon^{2mi} \lambda_i^m \\
      & \alert{\prec_e} \varepsilon^{3dm-2im} \cdot \lambda_i^m + \varepsilon^{mi} \lambda_i^m \\
      & \preccurlyeq_e \varepsilon^{2mi} \varepsilon^{mi} \lambda_i^m.
    \end{align*}
    Then there exists $m_2 > 0$ such that for every $m \geq m_2$, the condition $J(\frakA,i,m)$ holds.

    For condition $K(\frakA,i,m)$, we have
    \begin{align*}
      \deg_i(f^{2m}) \asymp_e \lambda_i^{2m} \alert{\succ_e} \varepsilon^{m} \cdot \lambda_i^{2m} \succcurlyeq_e \varepsilon^{3dm-2im} \cdot \lambda_i^m \cdot \lambda_i^m \succcurlyeq_e \varepsilon^{-mi} \Theta(\frakA; i)_m \cdot \deg_i(f^{m}).
    \end{align*}
    Hence there exists $m_3 > 0$ such that for every $m \geq m_3$, the condition $K(\frakA,i,m)$ holds.

    Then for every $m \geq \max\{m_1, m_2, m_3\}$, conditions $I(\frakA,i,m)$, $J(\frakA,i,m)$ and $K(\frakA,i,m)$ hold.
  \end{proof}

  \begin{lemma}\label{lem:growth_of_sigma_L_m}
    Suppose that $\mu_i > \mu_{i+1}$.
    The for every $\delta \in (0, 1)$ with $\delta \mu_i > \mu_{i+1}$, we have 
    \[ \Sigma(\mu_i \mu_{i-j} \delta; L_m^j)_m \asymp_e \mu_i^m \quad \text{ for every } j = 0, 1, \ldots, i-1. \]
  \end{lemma}
  \begin{proof}
    Similar to the proof of \cref{lemma:estimation_of_Sigma}.
    We have 
    \begin{align*}
      & M(\mu_i \mu_{i-j} \delta)_m^{d-j} \cdot L_m^{j} \\
      = & \sum_{k=0}^{d-j} \binom{d-j}{k} L_{2m}^{k} \cdot (\mu_{i-j}^m \mu_i^m \delta^m L)^{d-j-k} \cdot L_m^{j} \\
      \asymp_e & \sum_{k=0}^{d-j} \mu_1^{2m} \cdots \mu_k^{2m} \cdot \mu_{k+1}^m \cdots \mu_{k+j}^m \cdot \mu_{i-j}^{m(d-j-k)} \mu_i^{m(d-j-k)} \delta^{m(d-j-k)}.
    \end{align*}
    Set
    \[ a_k := \mu_1^2 \cdots \mu_k^2 \cdot \mu_{k+1} \cdots \mu_{k+j} \cdot \mu_i^{d-j-k} \mu_{i-j}^{d-j-k} \delta^{d-j-k}. \]
    Then, by the choice of $\delta$, we have
    \begin{equation}\label{eq:ratio_of_a_k_in_estimation_of_Sigma_and_setting_of_delta}
      \frac{a_{k+1}}{a_k} = \frac{\mu_{k+1} \cdot \mu_{k+1+j}}{\mu_{i-j} \cdot \mu_i} \cdot \delta^{-1} \begin{cases}
        > 1, & k \leq i-j-1, \\
        < 1, & k \geq i-j .
      \end{cases}
    \end{equation}
    Hence the maximum of $a_k$'s is attained at $k = i-j$.
    Then 
    \begin{equation}\label{eq:estimation_of_numerator_in_Sigma_L_m_in_lemma}
      M(\mu_i \mu_{i-j} \delta)_m^{d-j} \cdot L_m^{j} \asymp_e \mu_1^{2m} \cdots \mu_{i-j}^{2m} \cdot \mu_{i-j+1}^m \cdots \mu_{i}^m \cdot \mu_i^{m(d-i)} \mu_{i-j}^{m(d-i)} \delta^{m(d-i)}.
    \end{equation}

    Similarly, we have
    \begin{align*}
      M(\mu_i \mu_{i-j} \delta)_m^{d-j-1} \cdot L_m^{j+1} & \asymp_e \sum_{k=0}^{d-j-1} \mu_1^{2m} \cdots \mu_k^{2m} \cdot \mu_{k+1}^m \cdots \mu_{k+j+1}^m \cdot \mu_{i-j}^{m(d-j-1-k)} \mu_i^{m(d-j-1-k)} \delta^{m(d-j-1-k)}.
    \end{align*}
    Set 
    \[ b_k := \mu_1^2 \cdots \mu_k^2 \cdot \mu_{k+1} \cdots \mu_{k+j+1} \cdot \mu_i^{d-j-1-k} \mu_{i-j}^{d-j-1-k} \delta^{d-j-1-k}. \]
    Then we have
    \begin{equation}\label{eq:ratio_of_b_k_in_estimation_of_Sigma_and_setting_of_delta}
      \frac{b_{k+1}}{b_k} = \frac{\mu_{k+1} \cdot \mu_{k+1+j+1}}{\mu_{i-j} \cdot \mu_i} \cdot \delta^{-1} 
      \begin{dcases}
      \geq \dfrac{\mu_{k+1+1} \cdot \mu_{k+1+j+1}}{\mu_{i-j} \cdot \mu_i} \cdot \delta^{-1} = \frac{a_{k+2}}{a_{k+1}} > 1, & k < i-j-1; \\
      = \dfrac{\mu_{i-j} \cdot \mu_{i+1}}{\mu_{i-j} \cdot \mu_i} \cdot \delta^{-1} < 1, & k = i-j-1; \\
      \leq \dfrac{\mu_{k+1} \cdot \mu_{k+1+j}}{\mu_{i-j} \cdot \mu_i} \cdot \delta^{-1} = \frac{a_{k+1}}{a_k} < 1, & k \geq i-j.
    \end{dcases}
    \end{equation}
    Hence the maximum of $b_k$'s is attained at $k = i-j-1$.
    Then
    \begin{equation}\label{eq:estimation_of_denominator_in_Sigma_L_m_in_lemma}
      M(\mu_i \mu_{i-j} \delta)_m^{d-j-1} \cdot L_m^{j+1} \asymp_e \mu_1^{2m} \cdots \mu_{i-j-1}^{2m} \cdot \mu_{i-j}^m \cdots \mu_{i-1}^m \cdot \mu_i^{m(d-i)} \mu_{i-j}^{m(d-i)} \delta^{m(d-i)}.
    \end{equation}

    Combining \cref{eq:estimation_of_numerator_in_Sigma_L_m_in_lemma,eq:estimation_of_denominator_in_Sigma_L_m_in_lemma}, we get 
    \begin{align*}
      \Sigma(\mu_i \mu_{i-j} \delta; L_m^j)_m & = \frac{M(\mu_i \mu_{i-j} \delta)_m^{d-j} \cdot L_m^{j}}{M(\mu_i \mu_{i-j} \delta)_m^{d-j-1} \cdot L_m^{j+1}} \\
      & \asymp_e \frac{\mu_1^{2m} \cdots \mu_{i-j}^{2m} \cdot \mu_{i-j+1}^m \cdots \mu_{i}^m \cdot \mu_i^{m(d-i)} \mu_{i-j}^{m(d-i)} \delta^{m(d-i)}}{\mu_1^{2m} \cdots \mu_{i-j-1}^{2m} \cdot \mu_{i-j}^m \cdots \mu_{i-1}^m \cdot \mu_i^{m(d-i)} \mu_{i-j-1}^{m(d-i)} \delta^{m(d-i)}} \\
      & = \mu_{i-j}^{m}.
    \end{align*}
    We are done.
  \end{proof}
  \begin{remark}
    The \cref{eq:ratio_of_a_k_in_estimation_of_Sigma_and_setting_of_delta,eq:ratio_of_b_k_in_estimation_of_Sigma_and_setting_of_delta} are the places where we use the condition $\delta \mu_i > \mu_{i+1}$.
  \end{remark}
