\subsection{Grothendieck-Riemann-Roch theorem}


  Let $X$ be a scheme, $Z \subset X$ be a closed subscheme, and $U = X \setminus Z$ the open complement. 
  Then 
  \[ G_0(Z) \to G_0(X) \to G_0(U) \to 0 \]
  is exact.

\subsection{Topological filtration}

  We define a filtration 
  \[ 0 = F_{-1} G_0(X) \subset F_0 G_0(X) \subset F_1 G_0(X) \subset \cdots \subset F_{\dim X} G_0(X) = G_0(X) \]
  by letting $F_k G_0(X)$ be the subgroup generated by classes of coherent sheaves supported in dimension $\leq k$. 
  
  \begin{lemma}
    Let $X$ be a scheme.
    \begin{enumerate}
      \item For a coherent sheaf $\calF$ on $X$, 
        \[ [\calF] \equiv \sum_{V \subset X,\ \dim V = i} \ell_{\calO_{X,V}}(\calF_V) [\calO_V] \mod F_{i-1} G_0(X), \]
        where the sum is taken over all irreducible closed subschemes $V$ of dimension $i$.
        In particular, $F_i G_0(X)$ is generated by classes of the form $[\calO_V]$ for irreducible closed subschemes $V \subset X$ of dimension $\leq i$.
      \item Let $f:X \to Y$ be a proper morphism of schemes. 
        Then $f_* F_i G_0(X) \subset F_i G_0(Y)$ for all $i$.
        In particular, $\Gr_i G_0(X) := F_i G_0(X) / F_{i-1} G_0(X)$ is covariant functorial with respect to proper morphisms.
    \end{enumerate}
  \end{lemma}

\subsection{Riemann-Roch without denominators}

  Let $X$ be a smooth variety.
  Define 
  \[ \CH(X)^{\times} := \{ \sum_{i \geq 0} c_i | c_i \in \CH^i(X),\ c_0 = 1 \}. \]
  This is a multiplicative group.
  By Whitney sum formula, we get a group homomorphism 
  \[ c: K(X) \to \CH(X)^{\times}, \quad [\calE] \mapsto c(\calE) := 1 + c_1(\calE) + c_2(\calE) + \cdots. \]
  Let $f: X \to Y$ be a closed embedding of smooth varieties, $r = \codim_X Y$, $N = N_{X/Y}$ be the normal bundle.
  We want a formula for $c(f_* \calE)$ for vector bundles $\calE$ on $X$.

  Consider a model case:
  \[ Y = \bbP(N \oplus \calO), \quad X = Z(s) \subset Y, \]
  where $s$ is the canonical section of the universal quotient bundle $\calQ$ on $Y$ and $p: Y \to X$ is the projection.
  Consider $f: X \to N \to Y$.
  In this case, $f_*[E] = \sum_{i \geq 0} (-1)^i [\wedge^i \calQ^\vee \otimes p^* \calE]$ in $K(Y)$.
  Hence 
  \[ c(f_* \calE) = \prod_{i =0}^r c(\wedge^i \calQ^\vee \otimes p^* \calE)^{(-1)^i}. \]

  \begin{lemma}
    Fix non-negative integers $r, r'$.
    Then there exists $p \in \bbZ[[T_1, \ldots, T_r, U_1, \ldots, U_{r'}]]$ such that 
    for any smooth variety $X$, any vector bundles $\calD, \calE$ on $X$ of ranks $r', r$ respectively, 
    we have
    \[ \prod_{i=0}^r c(\wedge^i \calD^\vee \otimes \calE)^{(-1)^i} = 1 + c_r(\calD) \cdot P(\calD, \calE) \in \CH_*(X), \]
    where $P(\calD, \calE)$ is the specialization of $p$ at $T_i = c_i(\calD), U_j = c_j(\calE)$.
  \end{lemma}
