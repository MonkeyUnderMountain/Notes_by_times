\subsection{Higher Chow groups revisited}

\paragraph{Goal:} Show $X \to \CH^i(X,n)$ is a presheaf with transfers.

\begin{definition}
  Let $X, Y \in \Sm_\kk$, $\alpha \in \Cor(X,Y)$.
  Define $Z^i_\alpha(Y,n)$ to be the subgroup of $Z^i(Y,n)$ generated by all subvarieties $Z \subset Y \times \Delta^m$ such that $\alpha \times \id_{\Delta^n}$ meets $X \times Z$ properly in $X \times Y \times \Delta^n$ for all faces of $\Delta^m \injmap \Delta^n$.
  Then we can define $\alpha^* \colon Z^i_\alpha(Y,n) \to Z^i(X,n)$ 
\end{definition}
