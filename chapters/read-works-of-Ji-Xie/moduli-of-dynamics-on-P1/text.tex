\section{moduli-of-dynamics-on-P1}


\begin{definition}[Moduli space of rational maps on $\bbP^1$]
	Let $Rat_d = \bbP^{d} \times \bbP^d \setminus R$ be the space of rational maps of degree $d$ on $\bbP^1$
\end{definition}


\begin{theorem}
	The geometric quotient $Rat_d / \PGL_2$ exists and is an affine variety of dimension $2d - 2$.
\end{theorem}
