\section{PCF maps}

\begin{definition}[Post-critically finite maps]
	A endomorphism $f: \bbP^1 \to \bbP^1$ is called \emph{post-critically finite} (or \emph{PCF} for short) if the forward orbit of each critical point of $f$ is finite.
\end{definition}

\begin{theorem}
	Let $f: \bbP^1(\bbC) \to \bbP^1(\bbC)$ be an endomorphism of degree $d \geq 2$.
	Suppose that $f$ is PCF and is not a flexible Latt\`es map.
	Then the corresponding point $[f] \in \calM_d$ is defined over $\overline{\bbQ}$.
\end{theorem}


\begin{theorem}
	The set of PCF maps in $\calM_d$ is Zariski dense.
\end{theorem}


\begin{conjecture}[Dynamical Andr\'e-Oort conjecture]
	Let $f_t$ be a non-isotrivial algebraic family of endomorphisms of $\bbP^1$ of degree $d \geq 2$ parameterized by a variety $\Lambda$ of dimension $N$.
	TFAE:
	\begin{enumerate}
		\item the PCF points in $\Lambda$ are Zariski dense;
		\item there are at most $N$ dynamically independent critical points in the family $f_t$.
	\end{enumerate}
\end{conjecture}
