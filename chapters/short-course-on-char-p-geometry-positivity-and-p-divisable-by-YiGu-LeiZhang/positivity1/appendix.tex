\begin{theorem}[Cohomology and proper base change, \Yang{reference request}]
	Let $f:X\to Y$ be a proper morphism of noetherian schemes and $\calF$ a coherent sheaf on $X$ flat over $Y$.
	For any $y\in Y$, there exists a natural map
	\[ \phi^i_y: (R^if_*\calF)\otimes_{\calO_Y} k(y) \to H^i(X_y, \calF|_{X_y}). \]
	Suppose that $\phi^i_y$ is surjective for some $y\in Y$.
	Then we have
	\begin{enumerate}
		\item there exists an open neighborhood $U$ of $y$ such that $\phi^i_{y'}$ is an isomorphism for any $y'\in U$;
		\item $\phi^{i-1}_y$ is surjective if and only if $R^if_*\calF$ is locally free on some open neighborhood of $y$.
	\end{enumerate}
	\Yang{}
\end{theorem}
